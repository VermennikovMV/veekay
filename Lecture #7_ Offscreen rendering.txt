VkFormat candidates[] = { // Набрали распространенных кандидатов
	VK_FORMAT_D32_SFLOAT,
	VK_FORMAT_D32_SFLOAT_S8_UINT,
VK_FORMAT_D24_UNORM_S8_UINT,
};

os_depth_image_format = VK_FORMAT_UNDEFINED;

for (const auto& f : candidates) {
	VkFormatProperties properties;
	vkGetPhysicalDeviceFormatProperties(physical_device, f, &properties);

	// Если формат поддерживает оптимальное расположение в памяти GPU, то выбираем его
	if (properties.optimalTilingFeatures & 
         VK_FORMAT_FEATURE_DEPTH_STENCIL_ATTACHMENT_BIT) {
			os_depth_image_format = f;
			break;
	}
}

Нам нужны VkRenderPass, VkFramebuffer и VkPipeline
VkRenderPass расскажет, в каком формате будут изображения и когда
VkFramebuffer свяжет множество VkImageView с описанием VkRenderPass
VkPipeline при создании возьмёт VkRenderPass, чтобы знать какой формат изображений нужен для рисования

Но нам они больше не понадобятся! Теперь в Vulkan есть расширение Dynamic Rendering, которое “упрощает” работу с рисованием в изображения
Для Dynamic Rendering нам нужен всего один(!) VkPipeline, где будут описаны форматы изображений для рисования, привязка описывается потом, когда начнем рисовать
